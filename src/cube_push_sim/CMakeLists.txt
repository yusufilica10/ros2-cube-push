cmake_minimum_required(VERSION 3.10)
project(cube_push_sim)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosgraph_msgs REQUIRED)
add_executable(clock_throttle scripts/clock_throttle.cpp)
target_link_libraries(clock_throttle rclcpp::rclcpp ${rosgraph_msgs_TARGETS})
install(TARGETS clock_throttle DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch worlds urdf config maps DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/teleop_wasd.py DESTINATION lib/${PROJECT_NAME} RENAME teleop_wasd)
install(PROGRAMS scripts/cube_detector.py DESTINATION lib/${PROJECT_NAME} RENAME cube_detector)
install(FILES scripts/cube_geometry.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/velocity_mux.py DESTINATION lib/${PROJECT_NAME} RENAME velocity_mux)
install(PROGRAMS scripts/mission.py DESTINATION lib/${PROJECT_NAME} RENAME mission)
install(PROGRAMS scripts/evaluate_run.py DESTINATION lib/${PROJECT_NAME} RENAME evaluate_run)
install(PROGRAMS scripts/navigation_ready.py DESTINATION lib/${PROJECT_NAME} RENAME navigation_ready)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(perception_geometry test/test_cube_detector.py TIMEOUT 30)
endif()
ament_package()
